# =============================================================================
# GltfInstancingAndTilingTool 完整参数配置
# 包含所有可配置参数及说明，未设置的参数将使用默认值
# =============================================================================

# -----------------------------------------------------------------------------
# 一、基础配置
# -----------------------------------------------------------------------------
# 输入目录，包含待处理的 GLB 文件（必填）
input_directory = D:/Data/BIM/住宅标准层

# 输出目录（必填）
output_directory = D:/Experiments/output

# 是否将所有 GLB 合并为一个模型后再做实例化检测。默认 false
merge_all_glb = false

# 最小实例数阈值：实例数 ≥ 此值的组才归类为实例化组。默认 2
instance_limit = 3

# 是否允许非均匀缩放（xyz 不同）的 mesh 被归为同一实例组。默认 false
allow_non_uniform_scale_instancing = true

# 是否启用网格分割，为 Quadtree 生成按空间分割的 GLB。默认 false
mesh_segmentation = false

# CSV 目录，用于构件 ID 与 mesh 的映射（可选）
# csv_directory = D:/Data/csv

# -----------------------------------------------------------------------------
# 二、实例化检测（Stage 1 / semantic_material_geometric 模式）
# -----------------------------------------------------------------------------
# 实例化检测模式，当前仅支持 semantic_material_geometric
# instancing_detection_mode = semantic_material_geometric

# 语义分组字段，逗号分隔，如 category,family,type。空表示不做语义约束，所有 mesh 归入同一组。
# 仅写 category 则只按类别分组；category,family 则按类别+族分组。默认 category,family,type
semantic_hash_fields = category,family,type

# Stage 1 相似度阈值，逗号分隔，值域 [0,1]，越高越严格。默认 0.95,0.90,0.85,0.80,0.75
similarity_thresholds = 0.95,0.90,0.85,0.80,0.75

# HLOD/非实例化 LOD 单阈值模式（hlod_similarity_thresholds 为空时使用）。默认 0.70
hlod_similarity_threshold = 0.70

# Hausdorff 点云采样上限，0 表示不采样；越小越快但可能更不精确。默认 2000
hausdorff_max_sample_points = 2000

# 是否启用 ICP 点云配准：在 Hausdorff 比较前对齐旋转，处理 mesh 局部旋转差异。默认 false
# enable_icp_alignment = false

# 语义为 Unknown 的 mesh 是否允许跨 mesh 聚类。默认 false
allow_unknown_cross_mesh_clustering = false

# 材质过滤模式：none（不过滤）、hash（材质+纹理 hash 相同才比较）、index（材质索引相同才比较）。默认 none
material_filter_mode = none

# 语义数据路径：单文件或文件夹；若为文件夹，按 input_directory 下 GLB 文件名匹配同名 .RISCRVT（有语义数据时必填）
semantic_data_path = D:/Data/semantic

# 当 semantic_data_path 为文件夹时，用于匹配的 GLB 来源目录（默认用 input_directory）
# HLOD 使用 split/segmented 输入时，自动设为原始 input_directory，以正确匹配 RISCRVT
# semantic_input_directory = D:/Data/BIM/住宅标准层

# -----------------------------------------------------------------------------
# 三、Instance LOD（实例化 LOD）
# -----------------------------------------------------------------------------
# 是否启用实例化 LOD 生成。默认 false
enable_instance_lod_generation = true

# LOD 层级数（如 5 表示 LOD5→LOD1）。默认 5
lod_level_count = 5

# 屏幕空间误差，用于 tileset 的 geometricError。默认 16.0
target_screen_sse = 16.0

# 每级 LOD 的 Hausdorff 相似度阈值，LOD5→LOD1 逐步放宽，逗号分隔。默认 0.95,0.90,0.85,0.80,0.75
instance_lod_similarity_thresholds = 0.95,0.90,0.85,0.80,0.75

# Instance LOD 实例数阈值；-1 表示使用 Stage 1 的 instance_limit
instance_lod_instance_limit = -1

# Instance LOD 材质过滤：none / hash / index。默认 none
instance_lod_material_filter_mode = none

# LOD4 体积聚类回退（similarityThresholdsPerLevel 为空时使用）。默认 0.05
lod4_size_tolerance = 0.05

# LOD3 长宽比聚类回退。默认 0.20
lod3_aspect_ratio_tolerance = 0.20

# 是否启用语义检查（Family/Category 聚类）。默认 true
enable_semantic_check = true

# 是否启用几何检查（体积、长宽比等）。默认 true
enable_geometric_check = true

# -----------------------------------------------------------------------------
# 四、Non-instanced LOD（非实例化 LOD）
# -----------------------------------------------------------------------------
# 说明：实例化后处理沿用二节的 semantic_hash_fields、material_filter_mode、
# allow_unknown_cross_mesh_clustering、hausdorff_max_sample_points；
# 聚类时沿用三节的 lod4_size_tolerance、lod3_aspect_ratio_tolerance
# 是否对非实例化 mesh 做网格简化 LOD。默认 false
enable_non_instanced_lod_generation = false

# 非实例化 LOD 层级数。默认 3
non_instanced_lod_level_count = 3

# 每级简化的目标比例（如 0.5 表示保留约 50% 三角形）。默认 0.5
non_instanced_lod_ratio = 0.5

# 最小索引数，低于此值不再简化。默认 300
non_instanced_min_simplify_index_count = 300

# 非实例化 LOD 生成后，是否对每个 LOD 级再做实例化检测。默认 false
enable_non_instanced_lod_instancing = false

# 实例化检测后，是否再做 Family/Category 层级聚类（同 Instance LOD 的 LOD4/LOD3）。默认 false
enable_non_instanced_lod_clustering = false

# 非实例化 LOD 后处理实例化检测的每级相似度阈值，越粗糙越宽松。默认 0.95,0.90,0.85
non_instanced_lod_similarity_thresholds = 0.95,0.90,0.85

# 非实例化 LOD 实例化检测的实例数阈值；沿用六节的 hlod_instance_limit（未设置则用 instance_limit）

# -----------------------------------------------------------------------------
# 五、Quadtree / HLOD 配置
# -----------------------------------------------------------------------------
# 是否启用 Quadtree HLOD 流程。默认 false
enable_quadtree = true

# Quadtree 最大深度。默认 6
quadtree_max_depth = 6

# 每个 tile 最大对象数。默认 50
quadtree_max_objects_per_tile = 50

# -----------------------------------------------------------------------------
# 六、HLOD 实例化检测参数
# -----------------------------------------------------------------------------
# 说明：HLOD 实例化检测沿用二节的 semantic_hash_fields、material_filter_mode、
# allow_unknown_cross_mesh_clustering、hausdorff_max_sample_points
# 实例化检测后，是否再做 Family/Category 层级聚类（同 Instance LOD 的 LOD4/LOD3）。默认 false
enable_hlod_clustering = false

# HLOD 每层级相似度阈值，父 tile 越粗越宽松，逗号分隔。默认 0.70,0.65,0.60,0.55,0.50
hlod_similarity_thresholds = 0.70,0.65,0.60,0.55,0.50

# HLOD 实例数阈值；-1 表示使用 Stage 1 的 instance_limit
# 该值同时用于 Non-instanced LOD 实例化检测（enable_non_instanced_lod_instancing=true 时）
hlod_instance_limit = 3

# HLOD 是否允许非均匀缩放实例化。默认 false
hlod_allow_non_uniform_scale_instancing = true

# -----------------------------------------------------------------------------
# 七、实验模式配置
# -----------------------------------------------------------------------------
# 是否启用实验模式（新的文件夹结构与输出格式）。默认 false
enable_experiment_mode = true

# 是否使用符号链接节省空间（Windows 需管理员权限）。默认 false
use_symbolic_links = false

# 实验 1 文件夹名。默认 experiment1_baseline_comparison
# experiment1_name = experiment1_baseline_comparison

# 实验 2 文件夹名。默认 experiment2_lod_comparison
# experiment2_name = experiment2_lod_comparison

# 实验 3 文件夹名。默认 experiment3_mixed_hlod
# experiment3_name = experiment3_mixed_hlod

# 是否运行实验 6（跨 GLB HLOD 对比）。默认 false
run_cross_glb_hlod_experiment = true

# 数据集名称，用于组织输出文件夹
experiment_dataset_name = 多楼层建筑

# 策略 ID，用于实验 1 等对比实验（运行实验 1 时设置）
# experiment_strategy_id = 02_Moderate
